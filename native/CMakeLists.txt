cmake_minimum_required(VERSION 3.22)
project(MobileSpatialAINative VERSION 1.0.0 LANGUAGES CXX)
option(MSAI_BUILD_TESTS "Build native correctness tests" ON)
option(MSAI_BUILD_BENCHMARK "Build native microbenchmark" ON)
option(MSAI_ENABLE_FAST_MATH "Enable unsafe fast-math optimizations" OFF)
add_library(msai_core STATIC src/geometry.cpp src/frame_scheduler.cpp src/depth_postprocessor.cpp src/semantic_fusion.cpp)
target_include_directories(msai_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(msai_core PUBLIC cxx_std_17)
set_target_properties(msai_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
if(MSVC)
  target_compile_options(msai_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(msai_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(msai_core PRIVATE -O3)
  endif()
  if(MSAI_ENABLE_FAST_MATH)
    target_compile_options(msai_core PRIVATE -ffast-math)
  endif()
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT MSAI_IPO_SUPPORTED OUTPUT MSAI_IPO_ERROR)
if(MSAI_IPO_SUPPORTED AND CMAKE_BUILD_TYPE STREQUAL "Release")
  set_property(TARGET msai_core PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
if(NOT ANDROID AND MSAI_BUILD_TESTS)
  enable_testing()
  add_executable(msai_core_tests tests/test_core.cpp)
  target_link_libraries(msai_core_tests PRIVATE msai_core)
  add_test(NAME msai_core_tests COMMAND msai_core_tests)
endif()
if(NOT ANDROID AND MSAI_BUILD_BENCHMARK)
  add_executable(msai_benchmark bench/benchmark.cpp)
  target_link_libraries(msai_benchmark PRIVATE msai_core)
endif()
